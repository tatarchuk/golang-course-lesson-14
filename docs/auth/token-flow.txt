# Lesson 9 token flow against a fresh compose stack (the curl block of the README, run for real).
# Every command is printed before its response; "-> HTTP n" is the status code. Tokens are
# redacted to their first 24 characters (security_audit.md finding 6).

$ curl -s -X POST localhost:8080/api/v1/auth/register -H Content-Type: application/json -d {"email":"alice@example.com","password":"Correct-Horse-Battery-1"}

$ TOKEN=$(curl -s -X POST $BASE/auth/login -H 'Content-Type: application/json' -d '{"email":"alice@example.com","password":"Correct-Horse-Battery-1"}' | jq -r .token)
