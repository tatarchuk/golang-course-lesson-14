# awslocal only adds the endpoint. The same query with the plain AWS CLI pointed at Floci gives the same answer;
# without --endpoint-url the CLI goes to real AWS, which rejects the test credentials.
$ awslocal ec2 describe-instances --query "length(Reservations)"
1
$ aws --endpoint-url http://localhost:4566 ec2 describe-instances --query "length(Reservations)"
1
